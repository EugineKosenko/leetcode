* Система множин

Див. [[https://www.geeksforgeeks.org/introduction-to-disjoint-set-data-structure-or-union-find-algorithm/][статтю]].

#+begin_src rust :noweb-ref task1061:Визначення
  struct UnionFind { parent: Vec<usize> }

  impl UnionFind {
      fn new(size: usize) -> Self { UnionFind { parent: (0..size).collect() } }

      fn find(&mut self, x: usize) -> usize {
          if self.parent[x] != x { self.parent[x] = self.find(self.parent[x]); }
          self.parent[x]
      }

      fn union(&mut self, x: usize, y: usize) {
          let px = self.find(x);
          let py = self.find(y);
          if px < py { self.parent[py] = px; } else { self.parent[px] = py; }
      }
  }
#+end_src
