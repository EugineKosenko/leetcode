* Solution Search

Convert the source number to a string.

#+begin_src rust :noweb-ref task1432:Actions
  let s = num.to_string();
#+end_src

Find the first digit.

#+begin_src rust :noweb-ref task1432:Actions
  let d = s.chars().next().unwrap();
#+end_src

Iterate through all digit combinations, excluding the combination where ~d~ is replaced with ~0~, as such replacement
leads to leading zeros in the number.

#+begin_src rust :noweb-ref task1432:Actions
  let (mn, mx) = ('0'..='9').map(|d1| ('0'..='9').filter_map(move |d2| {
      if d1 == d && d2 == '0' { None } else { Some((d1, d2)) }
  })).flatten()
      .map(|(d1, d2)|
           s.chars().map(|d| if d == d1 { d2 } else { d })
           .collect::<String>().parse::<i32>().unwrap())
      .fold((i32::MAX, 0), |(mn, mx), n| (mn.min(n), mx.max(n)));
  mx - mn
#+end_src
