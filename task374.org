#+begin_src rust :noweb-ref task374:Визначення
  fn guess(num: i32, pick: i32) -> i32 {
      match num.cmp(&pick) {
          Ordering::Less => 1,
          Ordering::Equal => 0,
          Ordering::Greater => -1
      }
  }
#+end_src

* Цикл пошуку

При пошуку обчислюємо в циклі результат (~result~) як середнє значення між межами. При обчисленні
необхідно врахувати можливість переповнення, тому тимчасово обчислимо результат як беззнакове ціле.

Якщо результат вгадано, повертаємо його, інакше змінюємо відповідну межу інтервалу.

Оскільки результат завжди знаходиться всередині циклу, виконання після нього неможливе.

#+begin_src rust :noweb-ref task374:Дії
  loop {
      let result = ((high as u32 + low as u32) / 2) as i32;
      match guess(result, pick) {
          0 => { return result; },
          -1 => { high = result; },
          1 => { low = result; }
          c => panic!("Invalid guess result {}", c)
      }
  }
#+end_src
